name: Activate assistant document demand

on:
  workflow_dispatch:

permissions:
  contents: read

# Shares the existing release lock; no recurring job or additional Machine.
concurrency:
  group: CI and release-refs/heads/main
  cancel-in-progress: false

jobs:
  activate:
    if: github.ref == 'refs/heads/main' && vars.KOVA_CLOUDFLARE_BACKEND != '1'
    runs-on: ubuntu-latest
    timeout-minutes: 15
    environment: production
    steps:
      - uses: actions/checkout@fbc6f3992d24b796d5a048ff273f7fcc4a7b6c09 # v5
      - uses: superfly/flyctl-actions/setup-flyctl@ed8efb33836e8b2096c7fd3ba1c8afe303ebbff1 # v1
      - name: Activate only the accepted existing document host
        run: python3 scripts/activate_assistant_ingest.py --sha "$GITHUB_SHA"
        env:
          FLY_API_TOKEN: ${{ secrets.FLY_API_TOKEN }}
